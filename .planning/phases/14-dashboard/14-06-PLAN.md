    Create three thin React Query hook files. All three accept a `dateOnlyFilters: Pick<DashboardFilters, "range" | "date_from" | "date_to">` parameter — NEVER `fullFilters` (DASH-C2 prevention).

    `useTopPerforming.ts`:
    ```typescript
    import { useQuery } from "@tanstack/react-query";
    import { fetchTopPerforming } from "./api";
    import type { DashboardFilters, TopPerformingResponse } from "./types";

    export function useTopPerforming(filters: Pick<DashboardFilters, "range" | "date_from" | "date_to">) {
      return useQuery<TopPerformingResponse>({
        queryKey: ["dashboard", "top-performing", filters],
        queryFn: () => fetchTopPerforming(filters),
        staleTime: 5 * 60 * 1000,
        retry: 1,
      });
    }
    ```

    `useHighlights.ts` and `useYourStore.ts` follow the same pattern with `queryKey: ["dashboard", "highlights", filters]` and `["dashboard", "your-store", filters]` respectively. `useYourStore` takes an `enabled: boolean` parameter that maps to `useQuery({ enabled })` — so it only fetches when `isSingleShop=true`:

    ```typescript
    export function useYourStore(filters: ..., enabled: boolean) {
      return useQuery<YourStoreResponse>({
        queryKey: ["dashboard", "your-store", filters],
        queryFn: () => fetchYourStore(filters),
        staleTime: 5 * 60 * 1000,
        retry: 1,
        enabled,
      });
    }
    ```
