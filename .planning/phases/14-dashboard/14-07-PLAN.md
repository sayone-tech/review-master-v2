    Custom donut tooltip:
    ```tsx
    function CustomDonutTooltip({ active, payload, total }: { active?: boolean; payload?: any[]; total: number }) {
      if (!active || !payload?.length) return null;
      const { name, value } = payload[0];
      const pct = total > 0 ? Math.round(value/total*100) : 0;
      return (
        <div className="bg-white rounded-md shadow border border-line p-2 text-[14px]">
          <div className="capitalize font-semibold">{name}</div>
          <div>{value} reviews ({pct}%)</div>
        </div>
      );
    }
    ```
