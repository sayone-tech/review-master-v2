    ```python
    def org_admin_dashboard(request):
        from apps.reviews.selectors.reviews import get_accessible_shop_ids
        from apps.stores.models import Region, Shop

        user = request.user
        org_id = user.organisation_id
        accessible_shop_ids = list(get_accessible_shop_ids(user=user))

        shops = list(
            Shop.objects.filter(
                organisation_id=org_id, id__in=accessible_shop_ids,
            )
            .select_related("region")
            .values("id", "name", "region_id")
            .order_by("name")
        )
        region_ids = {s["region_id"] for s in shops if s["region_id"]}
        regions = list(
            Region.objects.filter(id__in=region_ids)
            .values("id", "name")
            .order_by("name")
        )
        is_single_shop = len(accessible_shop_ids) == 1

        context = {
            "regions_json": regions,
            "shops_json": shops,
            "is_single_shop": is_single_shop,
        }
        return render(request, "organisations/org_dashboard.html", context)
    ```
