- `ActionItemReadSerializer`: id, title, status, scope, priority, source, shop_id, shop_name (SerializerMethodField), assignee_id, assignee_name, due_date, created_at, updated_at, source_review_id, source_review (nested dict with id, comment[:200], rating, reviewer_name when source==AI else None), notes (nested ActionItemNoteSerializer many=True read_only=True — only used on retrieve, not list). - `ActionItemCreateSerializer`: title (5-200), scope (Shop/Brand), shop (PrimaryKeyRelatedField required when scope=SHOP, validated cross-field), priority (default Medium), assignee (optional), due_date (optional, must be today or future — `validators=[MinValueValidator(today)]`), initial_note (optional, max 2000). validate(): if scope=SHOP, require shop. if scope=BRAND, force shop=None.
- `ActionItemUpdateSerializer`: title, priority, due_date, assignee. Scope and shop are NOT in fields (read-only).
- `StatusTransitionSerializer`: status (ChoiceField over ActionItem.Status.choices, required).
- `ActionItemNoteCreateSerializer`: body (CharField, min_length=1, max_length=2000).

**Step 2 — FilterSet** (`apps/action_items/filters.py`):

```python
import django_filters
from typing import ClassVar
from django.db.models import Q
from apps.action_items.models import ActionItem


class ActionItemFilterSet(django_filters.FilterSet):
    shop = django_filters.NumberFilter(field_name="shop_id")
    status = django_filters.ChoiceFilter(choices=ActionItem.Status.choices)
    scope = django_filters.ChoiceFilter(choices=ActionItem.Scope.choices)
    assignee = django_filters.CharFilter(method="filter_assignee")
    from_date = django_filters.DateFilter(field_name="created_at", lookup_expr="gte")
    to_date = django_filters.DateFilter(field_name="created_at", lookup_expr="lte")
    search = django_filters.CharFilter(method="filter_search")
    review = django_filters.NumberFilter(field_name="source_review_id")

    def filter_assignee(self, qs, name, value):
        # value can be: "me" (resolved in view via request.user), "unassigned", or numeric id
        if value == "unassigned":
            return qs.filter(assignee__isnull=True)
        if value == "me":
            user = self.request.user if hasattr(self, "request") else None
            return qs.filter(assignee=user) if user else qs
        try:
            return qs.filter(assignee_id=int(value))
        except (TypeError, ValueError):
            return qs

    def filter_search(self, qs, name, value):
        return qs.filter(Q(title__icontains=value) | Q(notes__body__icontains=value)).distinct()

    class Meta:
        model = ActionItem
        fields: ClassVar[list[str]] = []
```

from apps.action_items.filters import ActionItemFilterSet
from apps.action_items.models import ActionItem
from apps.action_items.permissions import BrandScopeGuard
from apps.action_items.selectors.items import list_action_items
from apps.action_items.serializers import (
    ActionItemCreateSerializer,
    ActionItemListSerializer,
    ActionItemNoteCreateSerializer,
    ActionItemNoteSerializer,
    ActionItemReadSerializer,
    ActionItemUpdateSerializer,
    StatusTransitionSerializer,
)
from apps.action_items.services.lifecycle import (
    add_note,
    assign_action_item,
    create_action_item,
    transition_status,
)
from apps.common.pagination import DefaultPageNumberPagination  # adjust class name if different
from apps.common.permissions import IsOrgScoped


class ActionItemViewSet(
    mixins.ListModelMixin,
    mixins.RetrieveModelMixin,
    mixins.CreateModelMixin,
    mixins.UpdateModelMixin,
    viewsets.GenericViewSet,
):
    permission_classes: ClassVar[list] = [IsOrgScoped, BrandScopeGuard]
    pagination_class = DefaultPageNumberPagination
    filter_backends: ClassVar[list] = [DjangoFilterBackend, OrderingFilter]
    filterset_class = ActionItemFilterSet
    ordering_fields: ClassVar[list[str]] = ["created_at", "due_date", "status", "priority"]
    ordering = ["-created_at"]
    queryset = ActionItem.objects.none()

    def get_serializer_class(self):
        if self.action == "list":
            return ActionItemListSerializer
        if self.action == "create":
            return ActionItemCreateSerializer
        if self.action in ("update", "partial_update"):
            return ActionItemUpdateSerializer
        return ActionItemReadSerializer

    def get_queryset(self):
        user = self.request.user
        org_id = getattr(user, "organisation_id", None)
        if org_id is None:
            return ActionItem.objects.none()
        qs = list_action_items(organisation_id=org_id, user=user)
        if self.action == "retrieve":
            qs = qs.prefetch_related("notes__author")
        return qs

    def perform_create(self, serializer):
        data = serializer.validated_data
        item = create_action_item(
            organisation_id=self.request.user.organisation_id,
            title=data["title"],
            scope=data["scope"],
            priority=data.get("priority", ActionItem.Priority.MEDIUM),
            source=ActionItem.Source.MANUAL,
            actor=self.request.user,
            shop_id=data.get("shop").pk if data.get("shop") else None,
            assignee_id=data.get("assignee").pk if data.get("assignee") else None,
            due_date=data.get("due_date"),
            initial_note=data.get("initial_note"),
        )
        serializer.instance = item

    def perform_update(self, serializer):
        # Title/priority/due_date update path — direct field write, no AuditLog row
        # (only status, assignee, notes, creation get audit rows per ACTN-13).
        # If assignee changed, route through assign_action_item to write AuditLog.
        item = serializer.instance
        new_assignee = serializer.validated_data.get("assignee", "__unset__")
        serializer.save()
        if new_assignee != "__unset__":
            new_assignee_id = new_assignee.pk if new_assignee else None
            if new_assignee_id != item.assignee_id:
                assign_action_item(
                    action_item=item, assignee_id=new_assignee_id, actor=self.request.user
                )

    @action(detail=True, methods=["post"], url_path="transition-status")
    def transition_status_action(self, request, pk=None):
        item = self.get_object()
        s = StatusTransitionSerializer(data=request.data)
        s.is_valid(raise_exception=True)
        item = transition_status(
            action_item=item, new_status=s.validated_data["status"], actor=request.user
        )
        return Response(ActionItemReadSerializer(item).data)

    @action(detail=True, methods=["post"], url_path="add-note")
    def add_note_action(self, request, pk=None):
        item = self.get_object()
        s = ActionItemNoteCreateSerializer(data=request.data)
        s.is_valid(raise_exception=True)
        note = add_note(
            action_item=item, author=request.user, body=s.validated_data["body"]
        )
        return Response(ActionItemNoteSerializer(note).data, status=201)


@login_required
