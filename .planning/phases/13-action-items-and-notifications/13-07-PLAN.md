```tsx
type Props = {
  open: boolean;
  itemId: number | null;
  shops: ShopOption[];
  teamMembers: TeamMember[];
  onClose: () => void;
  onChanged: () => void;  // parent calls refetch()
};

export function ActionItemModal({ open, itemId, shops, teamMembers, onClose, onChanged }: Props) {
  const [item, setItem] = useState<ActionItemDetail | null>(null);
  const [activeTab, setActiveTab] = useState<"details" | "notes" | "source">("details");
  const [editing, setEditing] = useState(false);
  const [draft, setDraft] = useState<EditDraft>({ title: "", priority: "MEDIUM", due_date: "", assignee_id: null });
  const [saving, setSaving] = useState(false);
  const [error, setError] = useState<string | null>(null);

  // Reset state on open / itemId change
  useEffect(() => {
    if (!open || itemId == null) return;
    setActiveTab("details");
    setEditing(false);
    setError(null);
    void getActionItem(itemId).then(setItem);
  }, [open, itemId]);

  if (!open || !item) return null;

  const showSourceTab = item.source === "AI" && item.source_review;

  const handleStatusChange = async (newStatus: ActionItemStatus) => {
    try {
      const updated = await transitionStatus(item.id, newStatus);
      setItem(updated);
      emitToast({ kind: "success", title: `Status updated to ${STATUS_LABEL[newStatus]}` });
      onChanged();
    } catch {
      emitToast({ kind: "error", title: "Could not update status. Please try again." });
    }
  };

  const handleSave = async () => {
    setSaving(true);
    setError(null);
    try {
      const updated = await updateActionItem(item.id, {
        title: draft.title,
        priority: draft.priority,
        due_date: draft.due_date || null,
        assignee: draft.assignee_id,
      });
      setItem(updated);
      setEditing(false);
      emitToast({ kind: "success", title: "Changes saved" });
      onChanged();
    } catch (e) {
      setError(e instanceof Error ? e.message : "Save failed");
    } finally {
      setSaving(false);
    }
  };

  const handleAddNote = async (body: string) => {
    await addNote(item.id, body);
    const refreshed = await getActionItem(item.id);
    setItem(refreshed);
    emitToast({ kind: "success", title: "Note added" });
    onChanged();
  };

  return (
    <Modal open={open} onClose={onClose} size="lg" title={item.title}
           subtitle={item.scope === "BRAND" ? "Brand" : `Shop · ${item.shop_name ?? "—"}`}>
      <TabBar tabs={[
        { id: "details", label: "Details" },
        { id: "notes", label: "Notes" },
        ...(showSourceTab ? [{ id: "source", label: "Source Review" }] : []),
      ]} active={activeTab} onChange={setActiveTab} />

      {activeTab === "details" && (
        <DetailsTab item={item} editing={editing} draft={draft} setDraft={setDraft}
                    onEdit={() => {
                      setEditing(true);
                      setDraft({
                        title: item.title,
                        priority: item.priority,
                        due_date: item.due_date ?? "",
                        assignee_id: item.assignee_id ?? null,
                      });
                    }}
                    onCancelEdit={() => setEditing(false)}
                    onSave={handleSave}
                    onStatusChange={handleStatusChange}
                    saving={saving} error={error}
                    teamMembers={teamMembers} />
      )}
      {activeTab === "notes" && (
        <NotesTab notes={item.notes} onAddNote={handleAddNote} />
      )}
      {activeTab === "source" && showSourceTab && item.source_review && (
        <SourceReviewTab review={item.source_review} reviewId={item.source_review.id} />
      )}
    </Modal>
  );
}
```

**Step 5 — Backend: add `has_action_items` to `ReviewReadSerializer` via queryset annotation (B3)**.

Read `apps/reviews/serializers.py` first. If Phase 12 already exposes `has_action_items`, skip this step. Otherwise:

1. **Add the field via a queryset annotation in the Review viewset (NOT a SerializerMethodField with a sub-query).** Per RESEARCH.md Open Question 2, computing it from `obj.extracted_action_items` (already-loaded JSON) is acceptable but fragile — it conflates "Review has AI-suggested items in its enrichment payload" with "Review has actually-promoted ActionItem rows in DB". The semantically correct definition is "an ActionItem row exists with `source_review = review`". Use `Exists` so it stays a single SQL JOIN and the REVW-14 ≤5-query budget is preserved.

```python
# apps/reviews/views.py — get_queryset (or wherever ReviewReadSerializer's queryset is built)
from django.db.models import Exists, OuterRef
from apps.action_items.models import ActionItem

qs = qs.annotate(
    has_action_items=Exists(
        ActionItem.objects.filter(source_review=OuterRef("pk"))
    )
)
```

2. **In `apps/reviews/serializers.py`**, expose the annotation as a plain `BooleanField`:

```python
class ReviewReadSerializer(serializers.ModelSerializer):
    has_action_items = serializers.BooleanField(read_only=True)
    # ...existing fields...
    class Meta:
        model = Review
        fields: ClassVar[list[str]] = [..., "has_action_items"]
```

This produces zero extra queries because the annotation is folded into the existing list-query JOIN.

**Why not `SerializerMethodField`:** A method field that runs `obj.action_items.exists()` would issue one query per row (N+1 in the list endpoint). A method field reading `obj.extracted_action_items` (the JSON) would be free but semantically wrong — it returns True even when no ActionItem row was created (e.g., if promotion is still pending or failed).
